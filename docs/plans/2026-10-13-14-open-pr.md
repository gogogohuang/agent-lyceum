# run 結束時由 dispatcher 開 PR 實作計畫

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** task 要求開 PR 時，lead 在 `done` 寫 `open_pr: true`，run 以 `completed` 結束後由 dispatcher 在沙箱外把 lead 分支推上去並用 `gh pr create` 開 PR；沒有任何 agent 的沙箱被放寬。

**Architecture:** 新增 `src/open-pr.ts`（可注入指令執行器的純邏輯：標題、描述、push、`gh pr create`）。`src/run-session.ts` 在 `done` 被接受時讀 `meta.open_pr === true`，run 結束、`finishLead` 之後呼叫 `maybeOpenPr`，結果放進 `RunSummary.pr` 與 `state.pr_url`，由 `printRun` 顯示。`src/prompt.ts` 告訴 lead（與單獨 agent）何時才可以寫 `open_pr: true`。`src/doctor.ts` 提示缺 `gh`／`origin`。

**Tech Stack:** TypeScript ESM、Node `child_process`、`git`、`gh`、Vitest、Biome。不新增依賴。

**Spec:** `docs/specs/14-open-pr.md`（決議 1–10；10 為本計畫開工前補上）

## Global Constraints

- Node >=20，macOS 與 Linux；不新增依賴。
- 預設行為不變：task 沒要求開 PR（`done` 沒有 `open_pr: true`）時，不 push、不呼叫 `gh`，`state.json` 不多任何欄位。
- 只認 lead 的 `done`：`mailbox.ts` 本來就拒絕非 lead 的 `done`，所以非 lead 寄的信即使帶 `open_pr: true` 也不會被讀（要有測試鎖住）。`run --agent` 的單獨 agent 就是它那個一人團隊的 lead。程式、prompt、文件與測試都不得寫死任何 agent 名稱。
- 沙箱設定不變：Claude `allowUnsandboxedCommands: false`、Codex `-s workspace-write`；commit、push、`gh pr create` 只由 dispatcher 執行。
- 不碰使用者的 working tree 與目前分支；PR 只從 run 的 lead 分支 `agent-lyceum/<run-id>/<lead>` 開。這個分支只存在於有 lead worktree 的 run（`dispatcher.max_concurrent_runs > 1`，見 `docs/specs/07-concurrent-runs.md`）；沒有 lead worktree 的 run 要求開 PR 時**不開**，在 run 結果與 `notes` 說明原因與解法，不改 `outcome`。
- 只在 `outcome` 為 `completed` 且 lead 分支有檔案變更時開。push 或 `gh` 失敗不改變 `outcome`：分支保留，結果訊息印出可手動執行的指令。不自動 merge，不等 CI。
- PR 描述 = lead 的 `done` 內文原樣，結尾空一行後一行 `🤖 Opened by agent-lyceum`。dispatcher 自己不往描述或標題加任何 agent／角色名稱。
- 破壞相容性或新增行為時，同步更新 `README.md`、`README.zh-TW.md`、`docs/commands.md`、`docs/commands.zh-TW.md`，並通過 `npm run check:docs`（中英標題結構必須一致）。
- 每個 Task 結束時 `npm run typecheck`、`npm test`、`npm run lint` 都要通過。
- Commit 訊息結尾加：`Co-Authored-By: Claude Sonnet 5.5 <noreply@anthropic.com>`（執行時若系統提示給了別的 attribution，以系統提示為準）。發版不在這份計畫內。

```ts
import { execFileSync } from "node:child_process";
import path from "node:path";
import { atomicWrite } from "./fs-util.js";

/** Runs `git` or `gh` in `cwd`. Injected so tests never touch a network or a real remote. */
export type CommandRunner = (cmd: "git" | "gh", args: string[], cwd: string) => { ok: boolean; out: string };

/** The real thing: runs outside any agent sandbox, with the user's own git and gh credentials. */
export const realRunner: CommandRunner = (cmd, args, cwd) => {
  try {
    const out = execFileSync(cmd, args, { cwd, encoding: "utf8", stdio: ["ignore", "pipe", "pipe"], timeout: 120_000 });
    return { ok: true, out: out.trim() };
  } catch (e) {
    const err = e as { stderr?: string | Buffer; message: string };
    return { ok: false, out: String(err.stderr ?? err.message).trim() };
  }
};

/** The one line that says who opened the pull request. It names no agent and no role. */
export const PR_FOOTER = "🤖 Opened by agent-lyceum";

const TITLE_MAX = 72;
const BODY_MAX = 60_000; // GitHub rejects bodies over 65536 characters

/** First non-empty line of the task summary, cut to 72 characters. */
export function prTitle(taskSummary: string): string {
  const first = taskSummary.split(/\r?\n/).map((l) => l.trim()).find(Boolean) ?? "agent-lyceum task";
  return first.length > TITLE_MAX ? `${first.slice(0, TITLE_MAX - 1)}…` : first;
}

/** The lead's `done` report as written, then one footer line. */
export function prBody(doneBody: string): string {
  return `${doneBody.trim().slice(0, BODY_MAX)}\n\n${PR_FOOTER}\n`;
}

/** The repo's default branch: `origin/HEAD`, else what `gh` says, else `main`. */
export function defaultBase(cwd: string, run: CommandRunner): string {
  const sym = run("git", ["symbolic-ref", "--short", "refs/remotes/origin/HEAD"], cwd);
  const m = sym.ok ? /^origin\/(.+)$/.exec(sym.out) : null;
  if (m?.[1]) return m[1];
  const gh = run("gh", ["repo", "view", "--json", "defaultBranchRef", "--jq", ".defaultBranchRef.name"], cwd);
  return gh.ok && gh.out ? gh.out : "main";
}

export type OpenPrResult =
  | { status: "opened"; url: string }
  | { status: "failed"; step: "push" | "create" | "write"; reason: string; manual: string[] };

/** What happened to a requested pull request; `skipped` means it was asked for but the run could not support it. */
export type PrOutcome = OpenPrResult | { status: "skipped"; reason: string };

export interface OpenPrInput {
  /** The lead's worktree: where the branch is checked out. */
  cwd: string;
  branch: string;
  title: string;
  body: string;
  /** Where the description is written for `gh --body-file` (and for finishing by hand). */
  bodyFile: string;
  run: CommandRunner;
}

const firstLine = (s: string) => s.split(/\r?\n/).find((l) => l.trim())?.trim() ?? "no output";

/** Push the branch and open a pull request against the default branch. Never throws; says how to finish by hand when a step fails. */
export function openPullRequest(i: OpenPrInput): OpenPrResult {
  const base = defaultBase(i.cwd, i.run);
  atomicWrite(i.bodyFile, i.body);
  const createArgs = ["pr", "create", "--head", i.branch, "--base", base, "--title", i.title, "--body-file", i.bodyFile];
  const manualPush = `git -C ${i.cwd} push -u origin ${i.branch}`;
  const manualCreate = `gh pr create --head ${i.branch} --base ${base} --title ${JSON.stringify(i.title)} --body-file ${path.resolve(i.bodyFile)}`;

  const push = i.run("git", ["push", "-u", "origin", i.branch], i.cwd);
  if (!push.ok) return { status: "failed", step: "push", reason: firstLine(push.out), manual: [manualPush, manualCreate] };

  const created = i.run("gh", createArgs, i.cwd);
  if (!created.ok) return { status: "failed", step: "create", reason: firstLine(created.out), manual: [manualCreate] };
  const url = created.out.split(/\s+/).find((w) => /^https?:\/\//.test(w));
  if (!url) return { status: "failed", step: "create", reason: `gh printed no URL: ${firstLine(created.out)}`, manual: [manualCreate] };
  return { status: "opened", url };
}
```

```ts
/** A runner that records every call and answers like a healthy git + gh. */
function fakePr(over: Partial<Record<string, { ok: boolean; out: string }>> = {}) {
  const calls: { cmd: string; args: string[]; cwd: string }[] = [];
  const answers: Record<string, { ok: boolean; out: string }> = {
    "git symbolic-ref": { ok: true, out: "origin/main" },
    "git push": { ok: true, out: "" },
    "gh pr": { ok: true, out: "https://github.com/o/r/pull/9\n" },
    ...(over as Record<string, { ok: boolean; out: string }>),

  const run: CommandRunner = (cmd, args, cwd) => {
    calls.push({ cmd, args, cwd });
    return answers[`${cmd} ${args[0]}`] ?? { ok: false, out: "unexpected" };

  return { run, calls };
}
```

```ts
  /**
   * The lead asked for a pull request in `done`. It is opened from the lead's branch, by the dispatcher, outside any agent sandbox.
   * A request that cannot be honoured is reported and noted; it never changes how the run turned out.
   */
  private maybeOpenPr(endReason: EndReason, outcome: RunOutcome, rb: ResultBranch | undefined): PrOutcome | undefined {
    if (!this.openPrRequested) return undefined;
    const { say, log, state, runDir } = this;
    const skip = (reason: string): PrOutcome => {
      this.note(`A pull request was asked for but not opened: ${reason}`);
      return { status: "skipped", reason };
    };
    if (endReason !== "done" || outcome !== "completed") return skip(`the run did not end completed (${endReason}, outcome ${outcome}).`);
    if (!this.leadSpace) return skip("this run has no worktree of its own, so there is no branch to push. Set dispatcher.max_concurrent_runs: 2 (or more) in project.yaml.");
    if (!rb) return skip(`the work could not be committed on ${this.leadSpace.branch}.`);
    if (!rb.files) return skip(`${rb.branch} has no file changes.`);

    const result = openPullRequest({
      cwd: this.leadSpace.root,
      branch: rb.branch,
      title: prTitle(state.task_summary),
      body: prBody(this.doneMessage?.body ?? ""),
      bodyFile: path.join(runDir, "pr-body.md"),
      run: this.prRunner,
    });
    log("pull-request", { ...result });
    if (result.status === "opened") {
      state.pr_url = result.url;
      say(`  pull request opened: ${result.url}`);
    } else {
      this.note(`The pull request could not be opened (${result.step}: ${result.reason}). The branch ${rb.branch} is kept.${result.manual.length ? ` To finish by hand:\n${result.manual.map((m) => `  ${m}`).join("\n")}` : ""}`);
    }
    return result;
  }
```
